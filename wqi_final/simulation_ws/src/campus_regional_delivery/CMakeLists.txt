cmake_minimum_required(VERSION 3.8)
project(campus_regional_delivery)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
add_executable(regional_component_container src/regional_component_container.cpp)
target_compile_features(regional_component_container PUBLIC cxx_std_17)
ament_target_dependencies(regional_component_container rclcpp rclcpp_components lifecycle_msgs std_srvs nav2_core nav2_costmap_2d pluginlib)
target_link_libraries(regional_component_container ${CMAKE_DL_LIBS})
install(TARGETS regional_component_container DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config assets launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/regional_mission_manager scripts/regional_markers
  scripts/regional_localizer scripts/wait_for_regional_navigation scripts/regional_nav_container DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_regional_layout test/test_regional_layout.py TIMEOUT 120)
  ament_add_pytest_test(test_regional_fleet test/test_regional_fleet.py)
  ament_add_pytest_test(test_regional_mission test/test_regional_mission.py)
  ament_add_pytest_test(test_container_guard test/test_container_guard.py)
  ament_add_pytest_test(test_component_lifetime test/test_component_lifetime.py TIMEOUT 90)
endif()
ament_package()
